# A Meters view for the no-lyrics screen, drawn on a computer before it is built (planning/p4x-meters-view.md). Two programs
# over the synthetic music of ../speakers_options/scene.hpp:
#
#   meters_options  three variants of the view (an instrument after RME's displays: a third-octave spectrum with peak holds,
#                   level meters, a status line) through LVGL's software renderer into the panel's RGB565 buffer, at 1024 x 600,
#                   480 x 320 and 240 x 240, as BMP stills and frame sequences, or with "measure" a minute of each with the
#                   pixels LVGL flushed counted: the exact part of what a view costs.
#   meters_tune     no LVGL: spectrum.hpp's analysis held to a full-scale sine, to white noise and to one long FFT, and what one
#                   analysis costs on this computer.
#
#   cmake -S firmware/hearth-sink/preview/meters_options -B build -G Ninja -DCMAKE_BUILD_TYPE=Release \
#         -DLVGL_DIR=<firmware/hearth-sink/managed_components/lvgl__lvgl>
#   cmake --build build
#   build/meters_options <output directory> [measure]
#   build/meters_tune
#
# A project of its own, like ../speakers_options, and run by no CI job.
cmake_minimum_required(VERSION 3.28)
project(meters_options LANGUAGES C CXX)

set(LVGL_DIR "" CACHE PATH "lvgl 9.5 (managed_components/lvgl__lvgl of a P4 build)")
if(NOT EXISTS "${LVGL_DIR}/lvgl.h")
    message(FATAL_ERROR "set LVGL_DIR to an lvgl 9.5 checkout (no lvgl.h in '${LVGL_DIR}')")
endif()

set(HERE "${CMAKE_CURRENT_LIST_DIR}")
set(MAIN "${HERE}/../../main")
set(REPO "${HERE}/../../../..")

set(LV_BUILD_CONF_PATH "${HERE}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" lvgl EXCLUDE_FROM_ALL)

# How many separate rectangles LVGL keeps invalid in one refresh (its default is 32): past that it redraws the whole screen
# of the display. A spectrum has more rectangles than that in one 50 ms step; this is what is measured with 32, 128 and 256.
set(LV_INV_BUF 32 CACHE STRING "LV_INV_BUF_SIZE of the lvgl built here")
target_compile_definitions(lvgl PRIVATE LV_INV_BUF_SIZE=${LV_INV_BUF})

# libs/render's own export header is generated by its CMake, so spatial.cpp is built here against a stand-in written into
# this build's directory (not a file of the tree: the layering check reads every iclforge/render/export.hpp as the library's).
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shim/iclforge/render/export.hpp"
    "#pragma once\n#define ICLFORGE_RENDER_EXPORT\n")

set(INCLUDES
    "${CMAKE_CURRENT_BINARY_DIR}/shim"
    "${REPO}/libs/render/include"
    "${REPO}/libs/base/include"
    "${REPO}/libs/device/include"
    "${REPO}/libs/dsp/include")

add_executable(meters_options compare.cpp "${REPO}/libs/render/src/spatial.cpp")
target_compile_features(meters_options PRIVATE cxx_std_23)
target_include_directories(meters_options PRIVATE ${INCLUDES})
target_compile_definitions(meters_options PRIVATE "PREVIEW_FONT_DIR=\"${MAIN}/panel/fonts\"")
target_link_libraries(meters_options PRIVATE lvgl)

add_executable(meters_adaptive adaptive.cpp "${REPO}/libs/render/src/spatial.cpp")
target_compile_features(meters_adaptive PRIVATE cxx_std_23)
target_include_directories(meters_adaptive PRIVATE ${INCLUDES})
target_compile_definitions(meters_adaptive PRIVATE "PREVIEW_FONT_DIR=\"${MAIN}/panel/fonts\"")
target_link_libraries(meters_adaptive PRIVATE lvgl)

add_executable(meters_tune tune.cpp "${REPO}/libs/render/src/spatial.cpp")
target_compile_features(meters_tune PRIVATE cxx_std_23)
target_include_directories(meters_tune PRIVATE ${INCLUDES})
