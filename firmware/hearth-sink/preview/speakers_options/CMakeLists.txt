# The speakers view on a computer, before and after planning/p4x-speakers-options.md. Two programs, over the same
# synthetic music (scene.hpp: a groove panned onto 2.0, 5.1, 7.1.4 and 9.1.6, and for 7.1.4 objects that fly):
#
#   speakers_options  draws the speakers as they were (the 5 Hz glow, kept in compare.cpp), the responsive glow and the
#                     pulse style (both the real firmware/hearth-sink/main/panel/lvgl/now_screen.cpp) through LVGL's
#                     software renderer into the panel's RGB565 buffer, as BMP stills and frame sequences, or with
#                     "measure" a minute of each with the pixels LVGL flushed counted: the exact part of what a view costs.
#   speakers_tune     no LVGL: the glow's constants run over the same music, for choosing them (how many onsets show as a
#                     step, how often a steady level moves one, the pixels a second): room_plan.hpp's numbers came from it.
#
#   cmake -S firmware/hearth-sink/preview/speakers_options -B build -G Ninja -DCMAKE_BUILD_TYPE=Release \
#         -DLVGL_DIR=<firmware/hearth-sink/managed_components/lvgl__lvgl>
#   cmake --build build
#   build/speakers_options <output directory> [measure]
#   build/speakers_tune
#
# A project of its own, as ../CMakeLists.txt is of the lyric screen's pictures, and run by no CI job.
cmake_minimum_required(VERSION 3.28)
project(speakers_options LANGUAGES C CXX)

set(LVGL_DIR "" CACHE PATH "lvgl 9.5 (managed_components/lvgl__lvgl of a P4 build)")
if(NOT EXISTS "${LVGL_DIR}/lvgl.h")
    message(FATAL_ERROR "set LVGL_DIR to an lvgl 9.5 checkout (no lvgl.h in '${LVGL_DIR}')")
endif()

set(HERE "${CMAKE_CURRENT_LIST_DIR}")
set(MAIN "${HERE}/../../main")
set(REPO "${HERE}/../../../..")

set(LV_BUILD_CONF_PATH "${HERE}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" lvgl EXCLUDE_FROM_ALL)

# libs/render's own export header is generated by its CMake, so spatial.cpp is built here against a stand-in that is
# written into this build's directory (not a file of the tree: the layering check reads every iclforge/render/export.hpp
# as the library's).
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shim/iclforge/render/export.hpp"
    "#pragma once\n#define ICLFORGE_RENDER_EXPORT\n")

add_executable(speakers_options
    compare.cpp
    "${MAIN}/panel/lvgl/now_screen.cpp"
    "${REPO}/libs/render/src/spatial.cpp")
target_compile_features(speakers_options PRIVATE cxx_std_23)
target_include_directories(speakers_options PRIVATE
    "${MAIN}"
    "${CMAKE_CURRENT_BINARY_DIR}/shim"
    "${REPO}/libs/lyrics/include"
    "${REPO}/libs/render/include"
    "${REPO}/libs/base/include"
    "${REPO}/libs/device/include")
target_compile_definitions(speakers_options PRIVATE "PREVIEW_FONT_DIR=\"${MAIN}/panel/fonts\"")
target_link_libraries(speakers_options PRIVATE lvgl)

add_executable(speakers_tune tune.cpp "${REPO}/libs/render/src/spatial.cpp")
target_compile_features(speakers_tune PRIVATE cxx_std_23)
target_include_directories(speakers_tune PRIVATE
    "${CMAKE_CURRENT_BINARY_DIR}/shim"
    "${REPO}/libs/render/include"
    "${REPO}/libs/base/include")
